# Flask TTS Generation Route
@app.route('/generate', methods=['POST'])
def generate_audio():
  data = request.json
  voice_id = VOICES.get(data.get('voice'), VOICES['narrator'])
  url = f"https://api.elevenlabs.io/v1/text-to-speech/{voice_id}"
  response = requests.post(url, json={"text": data['text'], "model_id": "eleven_turbo_v2_5"}, headers=headers)
  return send_file(io.BytesIO(response.content), mimetype='audio/mpeg')
